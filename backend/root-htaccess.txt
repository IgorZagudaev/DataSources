RewriteEngine On
RewriteBase /DataSources/

# Если запрос к существующему файлу или папке - отдаём его
# (важно: проверка идёт ДО правила api/, иначе config-endpoint.php, permissions.php
#  и user-info.php перехватывались бы в api/index.php)
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# Остальные API запросы перенаправляем на api/index.php
RewriteRule ^api/(.*)$ api/index.php [QSA,L]

# Все остальные запросы - на index.html (SPA)
RewriteRule ^ index.html [L]
